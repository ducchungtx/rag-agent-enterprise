from rae_legal import RaeLegalClient

    api_key="rae_live_sec_9f8a...4b12",
    workspace_id="ws_phong_phap_che_01"

# Gửi yêu cầu truy vấn có kiểm chứng điều khoản
response = client.chat.create(
    model="rae-jurisprudence-v2",
    kb_id="kb_luat_dat_dai_2024",
    strict_citation=True,
    query="Điều kiện cấp Giấy chứng nhận quyền sử dụng đất theo Luật Đất đai mới?"
)

# In văn bản đã tổng hợp và danh mục trích dẫn chính thức

    print(f"[{cite.article_title}] Độ tin cậy: {cite.confidence_score}")
